cmake_minimum_required(VERSION 3.5.0)
project(HarmonySCP)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(DEFINED PACKAGE_FIND_FILE)
    include(${PACKAGE_FIND_FILE})
endif()
if(NOT OHOS_ARCH STREQUAL "arm64-v8a" AND NOT OHOS_ARCH STREQUAL "x86_64")
    message(FATAL_ERROR "Unsupported ABI: ${OHOS_ARCH}; use arm64-v8a or x86_64")
endif()
set(SSH_LIBS ${CMAKE_CURRENT_SOURCE_DIR}/../../../libs)
foreach(SSH_DEP libssh2.a libcrypto.a)
    if(NOT EXISTS "${SSH_LIBS}/${OHOS_ARCH}/${SSH_DEP}")
        message(FATAL_ERROR "Missing ${OHOS_ARCH}/${SSH_DEP}; see docs/validation/native-dependencies.md")
    endif()
endforeach()
add_library(entry SHARED napi_init.cpp ssh_client.cpp sftp_transfer.cpp terminal_session.cpp)
target_compile_definitions(entry PRIVATE HARMONYSCP_HILOG)
target_include_directories(entry PRIVATE ${SSH_LIBS}/include)
target_link_libraries(entry PUBLIC libace_napi.z.so libhilog_ndk.z.so ${SSH_LIBS}/${OHOS_ARCH}/libssh2.a ${SSH_LIBS}/${OHOS_ARCH}/libcrypto.a z)

add_subdirectory(termatlas)
